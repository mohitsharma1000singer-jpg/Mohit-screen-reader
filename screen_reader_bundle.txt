class AutoClickEditActivity : Activity() {

    private var index = -1
    private lateinit var nameBox: EditText
    private lateinit var stepsBox: EditText
    private lateinit var status: TextView

    private fun prefs() = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_autoclick_edit)
        nameBox = findViewById(R.id.edtMacroName)
        stepsBox = findViewById(R.id.edtMacroSteps)
        status = findViewById(R.id.txtMacroStatus)
        index = intent.getIntExtra("index", -1)

        val delete = findViewById<Button>(R.id.btnDeleteMacro)
        if (index >= 0) {
            val list = AutoClickStore.load(prefs())
            if (index < list.size) {
                nameBox.setText(list[index].name)
                stepsBox.setText(list[index].steps.joinToString("\n"))
            } else {
                index = -1
            }
        }
        if (index < 0) delete.isEnabled = false

        findViewById<Button>(R.id.btnSaveMacro).setOnClickListener { save() }
        delete.setOnClickListener { remove() }
    }

    private fun tell(message: String, thenFinish: Boolean) {
        status.text = message
        status.announceForAccessibility(message)
        if (thenFinish) Handler(Looper.getMainLooper()).postDelayed({ finish() }, 1800)
    }

    private fun save() {
        val name = nameBox.text.toString().trim()
        val steps = stepsBox.text.toString().split("\n").map { it.trim() }.filter { it.isNotEmpty() }
        if (name.isEmpty()) {
            tell("Please give the function a name.", false)
            return
        }
        if (steps.isEmpty()) {
            tell("Please paste at least one item to click.", false)
            return
        }
        val list = AutoClickStore.load(prefs())
        if (index >= 0 && index < list.size) {
            list[index] = Macro(name, steps)
        } else {
            list.add(Macro(name, steps))
        }
        AutoClickStore.save(prefs(), list)
        tell("Saved " + name + " with " + steps.size + (if (steps.size == 1) " step." else " steps."), true)
    }

    private fun remove() {
        val list = AutoClickStore.load(prefs())
        if (index >= 0 && index < list.size) {
            val name = list[index].name
            list.removeAt(index)
            AutoClickStore.save(prefs(), list)
            tell("Deleted " + name + ".", true)
        }
    }

import android.content.SharedPreferences
import org.json.JSONArray
import org.json.JSONObject

// Items of the Screen Reader Menu and the reading controls that can be switched on or off.
// The user stores only what is switched OFF, so new items added in future updates
// appear automatically (switched on).
object MenuConfig {
    val menuItems: List<Pair<String, String>> = listOf(
        "copy" to "Copy",
        "append" to "Append, add to what was copied before",
        "cut" to "Cut",
        "paste" to "Paste",
        "select" to "Select",
        "clear" to "Clear edit box",
        "ai" to "AI menu",
        "extensions" to "Extensions"
    )

    val readingItems: List<Pair<String, String>> = listOf(
        "character" to "Character",
        "word" to "Word",
        "line" to "Line",
        "paragraph" to "Paragraph",
        "actions" to "Actions",
        "slider" to "Slider"
    )

    fun disabled(prefs: SharedPreferences, kind: String): Set<String> =
        prefs.getStringSet("disabled_" + kind, emptySet()) ?: emptySet()
}

// Built-in extensions. Anything added here (or any auto click function the user creates)
// automatically shows up in the list of functions a gesture can run.
object Extensions {
    val builtin: List<Pair<String, String>> = listOf(
        "yt_skip" to "Skip ads on YouTube"
    )
}

class Macro(var name: String, var steps: List<String>)

// Auto click functions: a name plus a list of on-screen items to click, in order.
object AutoClickStore {
    fun load(prefs: SharedPreferences): MutableList<Macro> {
        val out = mutableListOf<Macro>()
        val s = prefs.getString("autoclick_functions", null) ?: return out
        try {
            val arr = JSONArray(s)
            for (i in 0 until arr.length()) {
                val o = arr.getJSONObject(i)
                val stepsArr = o.getJSONArray("steps")
                val steps = mutableListOf<String>()
                for (k in 0 until stepsArr.length()) steps.add(stepsArr.getString(k))
                out.add(Macro(o.getString("name"), steps))
            }
        } catch (e: Exception) {
            // ignore damaged data
        }
        return out
    }

    fun save(prefs: SharedPreferences, list: List<Macro>) {
        val arr = JSONArray()
        for (m in list) {
            val o = JSONObject()
            o.put("name", m.name)
            val st = JSONArray()
            for (x in m.steps) st.put(x)
            o.put("steps", st)
            arr.put(o)
        }
        prefs.edit().putString("autoclick_functions", arr.toString()).apply()
    }
}

import android.app.Activity

import android.content.Intent
import android.content.pm.ApplicationInfo
import android.os.Bundle

import android.widget.ListView
import android.widget.TextView

class GamingAppsActivity : Activity() {

    private class Row(val pkg: String, val label: String, val auto: Boolean)

    private val rows = mutableListOf<Row>()
    private lateinit var list: ListView
    private lateinit var status: TextView

    private fun prefs() = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_gaming_apps)
        list = findViewById(R.id.listApps)
        status = findViewById(R.id.txtGamingStatus)
        status.text = "Loading apps..."

        Thread {
            val pm = packageManager
            val intent = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
            val seen = HashSet<String>()
            val found = mutableListOf<Row>()
            for (ri in pm.queryIntentActivities(intent, 0)) {
                val pkg = ri.activityInfo.packageName
                if (pkg == packageName || !seen.add(pkg)) continue
                val label = ri.loadLabel(pm).toString()
                val auto = try {
                    pm.getApplicationInfo(pkg, 0).category == ApplicationInfo.CATEGORY_GAME
                } catch (e: Exception) {
                    false
                }
                found.add(Row(pkg, label, auto))
            }
            found.sortBy { it.label.lowercase() }
            runOnUiThread { show(found) }
        }.start()
    }

    private fun show(found: List<Row>) {
        rows.clear()
        rows.addAll(found)
        val p = prefs()
        val on = p.getStringSet("gaming_on", emptySet()) ?: emptySet()
        val off = p.getStringSet("gaming_off", emptySet()) ?: emptySet()

        val labels = rows.map { if (it.auto) it.label + ", detected as a game" else it.label }
        val adapter = ArrayAdapter(this, android.R.layout.simple_list_item_multiple_choice, labels)
        list.choiceMode = ListView.CHOICE_MODE_MULTIPLE
        list.adapter = adapter
        rows.forEachIndexed { i, r ->
            val checked = on.contains(r.pkg) || (r.auto && !off.contains(r.pkg))
            list.setItemChecked(i, checked)
        }
        list.setOnItemClickListener { _, _, position, _ ->
            val r = rows[position]
            val nowChecked = list.isItemChecked(position)
            val sp = prefs()
            val newOn = HashSet(sp.getStringSet("gaming_on", emptySet()) ?: emptySet())
            val newOff = HashSet(sp.getStringSet("gaming_off", emptySet()) ?: emptySet())
            if (nowChecked) {
                newOn.add(r.pkg)
                newOff.remove(r.pkg)
            } else {
                newOff.add(r.pkg)
                newOn.remove(r.pkg)
            }
            sp.edit().putStringSet("gaming_on", newOn).putStringSet("gaming_off", newOff).apply()
        }
        status.text = "Tick the apps that should use gaming mode. Apps detected as games are already ticked; untick one to turn it off."
    }
}

class GestureSettingsActivity : Activity() {

    private val rows = mutableListOf<String>()

    private fun prefs() = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_gesture_settings)
        list = findViewById(R.id.listGestures)
        adapter = ArrayAdapter(this, android.R.layout.simple_list_item_1, rows)
        list.adapter = adapter
        refreshRows()

        list.setOnItemClickListener { _, _, position, _ -> chooseFunction(position) }
        findViewById<Button>(R.id.btnResetGestures).setOnClickListener {
            val e = prefs().edit()
            for (g in GestureConfig.gestures) e.remove("gesture_" + g.first)
            e.apply()
            refreshRows()
            tell("All gestures were reset to their defaults.")
        }
    }

    private fun refreshRows() {
        rows.clear()
        for (g in GestureConfig.gestures) {
            rows.add(g.second + ": " + GestureConfig.functionLabel(GestureConfig.functionFor(prefs(), g.first)))
        }
        adapter.notifyDataSetChanged()
    }

    // Announce a little later so it is not cut off by the screen reader reading the screen again.
    private fun tell(message: String) {
        Handler(Looper.getMainLooper()).postDelayed({ list.announceForAccessibility(message) }, 900)
    }

    private fun chooseFunction(position: Int) {
        val gesture = GestureConfig.gestures[position]
        val current = GestureConfig.functionFor(prefs(), gesture.first)
        val funcs = GestureConfig.allFunctions(prefs())
        val labels = funcs.map { it.second }.toTypedArray()
        AlertDialog.Builder(this)
            .setTitle(gesture.second + ". Choose what it does")
            .setItems(labels) { _, which ->
                val chosen = funcs[which].first
                if (current == "menu" && chosen != "menu" && !menuMappedElsewhere(gesture.first)) {
                    tell("Not changed. Give the Screen reader menu to another gesture first, otherwise you could not open it.")
                } else {
                    prefs().edit().putString("gesture_" + gesture.first, chosen).apply()
                    refreshRows()
                    tell(gesture.second + " now does: " + GestureConfig.functionLabel(chosen))
                }
            }
            .show()
    }

    private fun menuMappedElsewhere(exceptKey: String): Boolean {
        return GestureConfig.gestures.any {

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        event ?: return
        when (event.eventType) {
            AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED -> {
                // The on-screen keyboard is its own window - it must not reset our focus.
                if (isInputMethodEvent(event)) return
                updateGamingMode(event.packageName?.toString())
                // While the menu is open, or right after we touch the clipboard
                // (Android 13 shows a system "copied" popup), ignore this so the
                // menu / focus is not reset.
                if (mode != Mode.NORMAL || System.currentTimeMillis() < suppressWindowEventsUntil) return
                if (gamingActive) return
                scheduleAutoFocus(event.text.joinToString(" ").trim())
            }
            AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED -> {
                if (gamingActive) speakGameText(event)
            }
            AccessibilityEvent.TYPE_VIEW_TEXT_CHANGED -> {
                if (mode == Mode.NORMAL && !gamingActive &&
                    System.currentTimeMillis() >= suppressWindowEventsUntil) echoTyping(event)
            }
            AccessibilityEvent.TYPE_ANNOUNCEMENT -> {
                // Apps (like YouTube) announce things such as a new slider position this way.
                if (mode != Mode.NORMAL) return
                val said = event.text.joinToString(" ").trim()
                if (said.isNotEmpty()) {
                    if (System.currentTimeMillis() < sliderWatchUntil) sliderSpoken = true
                    speak(said)
                }
            }
            AccessibilityEvent.TYPE_TOUCH_INTERACTION_START -> {
                touching = true
                if (voiceActive) stopVoiceTyping()
            }
            AccessibilityEvent.TYPE_TOUCH_INTERACTION_END -> {
                touching = false
            }
            AccessibilityEvent.TYPE_VIEW_HOVER_ENTER -> {
                if (mode != Mode.NORMAL) {
                    hintModalEntry()
                    return
                }
                handleHover(event)
            }
        }
    }

    /**
     * Explore-by-touch: when a finger is dragged (not swiped) directly onto a
     * screen item, Android's touch-exploration layer (enabled via
     * flagRequestTouchExplorationMode in accessibility_service_config.xml)
     * automatically hit-tests and reports the exact view under the finger as
     * a HOVER_ENTER event - no manual coordinate math needed here. This is
     * on top of (not instead of) swipe navigation and double-tap activation.
     */
    private fun handleHover(event: AccessibilityEvent) {
        val raw = event.source ?: return
        // If the finger is on a piece of text inside a clickable row, treat the
        // whole row as the item (same as swipe navigation does).
        var node: AccessibilityNodeInfo = raw
        if (!node.isClickable && !isSlider(node)) {
            var p = node.parent
            var d = 0
            while (p != null && d < 4) {
                if (p.isClickable && labelOf(p) != null) {
                    node = p
                    break
                }
                p = p.parent
                d++
            }
        }
        if (!node.isVisibleToUser) return
        if (labelOf(node) == null && !isSlider(node)) return

        focusedNode = node
        resetFocusState()
        // Best-effort: keep swipe navigation picking up from near where the
        // finger just was.
        val hovered = node
        val matchIndex = nodeList.indexOfFirst { it == hovered }
        if (matchIndex >= 0) currentIndex = matchIndex

        node.performAction(AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS)
        playSound(soundFocus)
        speak(spokenLabel(node))
    }

    private fun rebuildNodeList() {
        nodeList.clear()
        val root = rootInActiveWindow
        if (root != null) collectNodes(root, nodeList, false)
        // The on-screen keyboard is a separate window: include its keys after the app's items.
        try {
            for (w in windows) {
                if (w.type == AccessibilityWindowInfo.TYPE_INPUT_METHOD) {
                    val r = w.root
                    if (r != null) collectNodes(r, nodeList, false)
                }
            }

            // Keyboard keys are optional.
        }
    }

    // ---- Labels: one item per row, never an empty "item" followed by its own text ----

    private fun ownLabel(node: AccessibilityNodeInfo): String? {
        return node.text?.toString()?.takeIf { it.isNotBlank() }
            ?: node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
    }

    // Text of the non-clickable descendants of a clickable row, joined together.
    private fun gatherText(node: AccessibilityNodeInfo, out: MutableList<String>, depth: Int) {
        if (depth > 6 || out.size > 8) return
        for (i in 0 until node.childCount) {
            val c = node.getChild(i) ?: continue
            if (!c.isVisibleToUser) continue
            if (c.isClickable) continue // a separate control, it is its own item
            val own = ownLabel(c)
            if (own != null) out.add(own) else gatherText(c, out, depth + 1)
        }
    }

    private fun mergedText(node: AccessibilityNodeInfo): String? {
        if (!node.isClickable || node.isScrollable) return null
        val parts = mutableListOf<String>()
        gatherText(node, parts, 0)
        if (parts.isEmpty() || parts.size > 8) return null
        return parts.joinToString(", ")
    }

    // Full spoken label: its own text plus the text of plain children (e.g. a quick-settings tile
    // whose name and "On/Off" are separate views) plus its state (On/Off, checked).
    private fun labelOf(node: AccessibilityNodeInfo): String? {
        val parts = mutableListOf<String>()
        val own = ownLabel(node)
        if (own != null) parts.add(own)
        if (node.isClickable) {
            // Tiles often keep the name in the description and only the extra info in the text.
            val cd = node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
            if (cd != null && parts.none { it.contains(cd, true) || cd.contains(it, true) }) parts.add(0, cd)
            try {
                val lb = node.labeledBy?.let { ownLabel(it) }
                if (lb != null && parts.none { it.contains(lb, true) }) parts.add(0, lb)
            } catch (e: Exception) {
                // optional
            }
        }
        if (node.isClickable && !node.isScrollable && (own == null || own.length <= 40)) {
            val g = mutableListOf<String>()
            gatherText(node, g, 0)
            if (g.size <= 8) {
                for (x in g) if (parts.none { it.contains(x, true) }) parts.add(x)
            }
        }
        if (parts.isEmpty()) return null
        if (!isSlider(node)) {
            var state: String? = null
            if (Build.VERSION.SDK_INT >= 30) {
                state = node.stateDescription?.toString()?.takeIf { it.isNotBlank() }
            }
            if (state == null && node.isCheckable) {
                val cls = node.className?.toString() ?: ""
                state = if (cls.contains("Switch") || cls.contains("Toggle")) {
                    if (node.isChecked) "On" else "Off"
                } else {
                    if (node.isChecked) "checked" else "not checked"
                }
            }
            val st = state
            if (st != null && parts.none { it.contains(st, true) }) parts.add(st)
        }
        return parts.joinToString(", ")
    }

    private fun hasLabeledDescendant(node: AccessibilityNodeInfo, depth: Int = 0): Boolean {
        if (depth > 6) return false
        for (i in 0 until node.childCount) {
            val c = node.getChild(i) ?: continue
            if (!c.isVisibleToUser) continue
            if (ownLabel(c) != null) return true
            if (hasLabeledDescendant(c, depth + 1)) return true
        }
        return false
    }

    private fun collectNodes(node: AccessibilityNodeInfo?, out: MutableList<AccessibilityNodeInfo>, insideMerged: Boolean) {
        if (node == null) return
        if (!node.isVisibleToUser) return
        val label = labelOf(node)
        val clickableOk = node.isClickable && (label != null || !hasLabeledDescendant(node))
        val slider = isSlider(node)
        val eligible = clickableOk || slider || (!insideMerged && label != null)
        if (eligible) out.add(node)
        // Once a labelled clickable row has been added, its plain text children are
        // already part of its label: only real controls inside it stay separate items.
        val merged = insideMerged || (node.isClickable && label != null && eligible)
        for (i in 0 until node.childCount) {
            collectNodes(node.getChild(i), out, merged)
        }
    }

    // ---- Keeping the swipe list fresh (feeds like YouTube load items after the screen opens) ----

    private fun isDescendantOf(node: AccessibilityNodeInfo, ancestor: AccessibilityNodeInfo): Boolean {
        var cur: AccessibilityNodeInfo? = node.parent
        var d = 0
        while (cur != null && d < 20) {
            if (cur == ancestor) return true
            cur = cur.parent
            d++
        }
        return false
    }

    private fun boundsOf(node: AccessibilityNodeInfo): Rect {
        val r = Rect()
        node.getBoundsInScreen(r)
        return r
    }

    // Rebuilds the list from the live screen and finds the currently focused item in it again.
    private fun syncList(): Boolean {
        val old = focusedNode
        val oldLabel = old?.let { labelOf(it) }
        val oldTop = old?.let { boundsOf(it).top } ?: 0
        rebuildNodeList()
        if (old != null) {
            var idx = nodeList.indexOfFirst { it == old }
            if (idx < 0 && oldLabel != null) {
                var best = -1
                var bestDiff = Int.MAX_VALUE
                nodeList.forEachIndexed { i, n ->
                    if (labelOf(n) == oldLabel) {
                        val diff = Math.abs(boundsOf(n).top - oldTop)
                        if (diff < bestDiff) {
                            bestDiff = diff
                            best = i
                        }
                    }
                }
                idx = best
            }
            if (idx >= 0) {
                currentIndex = idx
                focusedNode = nodeList[idx]
                return true
            }
        }
        currentIndex = if (nodeList.isEmpty()) -1 else minOf(currentIndex, nodeList.size - 1)
        return false
    }

    private fun findScrollableAncestor(node: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
        var cur: AccessibilityNodeInfo? = node
        var d = 0
        while (cur != null && d < 25) {
            if (cur.isScrollable) return cur
            cur = cur.parent
            d++
        }
        return null
    }

    private fun findAnyScrollable(node: AccessibilityNodeInfo?, depth: Int = 0): AccessibilityNodeInfo? {
        if (node == null || depth > 25) return null
        if (node.isVisibleToUser && node.isScrollable) return node
        for (i in 0 until node.childCount) {
            val f = findAnyScrollable(node.getChild(i), depth + 1)
            if (f != null) return f
        }
        return null
    }

        if (mode != Mode.NORMAL) {
            // Inside the menu the gestures are fixed so you can never get stuck.
            when (gestureId) {
                AccessibilityService.GESTURE_SWIPE_RIGHT -> stepModal(1)
                AccessibilityService.GESTURE_SWIPE_LEFT -> stepModal(-1)
                AccessibilityService.GESTURE_SWIPE_DOWN_AND_RIGHT -> closeMenu()
                AccessibilityService.GESTURE_SWIPE_DOWN_AND_LEFT -> goBack()
            }
            return true
        }
        val key = gestureKey(gestureId) ?: return super.onGesture(gestureId)
        val prefs = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
        runFunction(GestureConfig.functionFor(prefs, key))
        return true
    }

    private fun globalAction(action: Int) {
        if (performGlobalAction(action)) playSound(soundActivate) else fail("Could not do that")
    }

    // Copy, cut, paste etc. on the focused item, straight from a gesture.
    private fun quickOp(op: (AccessibilityNodeInfo) -> OpResult) {
        val node = focusedNode ?: nodeList.getOrNull(currentIndex)
        if (node == null) {
            fail("Focus an item first")
            return
        }
        val r = op(node)
        if (r.ok) {
            playSound(soundActivate)
            speak(r.msg)
        } else {
            fail(r.msg)
        }
    }

    private fun scrollFromGesture(dir: Int) {
        val from = focusedNode ?: nodeList.getOrNull(currentIndex)
        val scroller = findVerticalScroller(from, dir) ?: findAnyScrollable(rootInActiveWindow)
        if (scroller != null && scrollList(scroller, dir)) {
            playSound(soundFocus)
        } else {
            fail(if (dir > 0) "Cannot scroll forward" else "Cannot scroll back")
        }
    }

    private fun runFunction(id: String) {
        when (id) {
            "none" -> {}
            "next" -> moveNext()
            "previous" -> movePrevious()
            "activate" -> activateCurrent()
            "read_next" -> readStep(1)
            "read_prev" -> readStep(-1)
            "control_next" -> changeReadingControl(1)
            "control_prev" -> changeReadingControl(-1)
            "menu" -> openMenu()
            "back" -> globalAction(GLOBAL_ACTION_BACK)
            "home" -> globalAction(GLOBAL_ACTION_HOME)
            "recents" -> globalAction(GLOBAL_ACTION_RECENTS)
            "notifications" -> globalAction(GLOBAL_ACTION_NOTIFICATIONS)
            "quick_settings" -> globalAction(GLOBAL_ACTION_QUICK_SETTINGS)
            "power_dialog" -> globalAction(GLOBAL_ACTION_POWER_DIALOG)
            "lock_screen" -> if (Build.VERSION.SDK_INT >= 28) globalAction(GLOBAL_ACTION_LOCK_SCREEN) else fail("Needs Android 9 or newer")
            "screenshot" -> if (Build.VERSION.SDK_INT >= 28) globalAction(GLOBAL_ACTION_TAKE_SCREENSHOT) else fail("Needs Android 9 or newer")
            "copy" -> quickOp { opCopy(it, false) }
            "append" -> quickOp { opCopy(it, true) }
            "cut" -> quickOp { opCut(it) }
            "paste" -> quickOp { opPaste(it) }
            "clear_edit" -> quickOp { opClear(it) }
            "voice_typing" -> beginVoiceFlow()
            "describe_screen" -> runAi(false)
            "describe_image" -> {
                menuTarget = focusedNode ?: nodeList.getOrNull(currentIndex)
                runAi(true)
            }
            "scroll_forward" -> scrollFromGesture(1)
            "scroll_backward" -> scrollFromGesture(-1)
            "repeat" -> if (lastSpoken.isNotEmpty()) speak(lastSpoken) else playSound(soundAlert)
            "stop_speech" -> tts?.stop()
            "volume_up" -> adjustAccessibilityVolume(true)
            "volume_down" -> adjustAccessibilityVolume(false)
            else -> runDynamicFunction(id)
        }
    }

    // Extensions and auto click functions chosen for a gesture.
    private fun runDynamicFunction(id: String) {
        if (id.startsWith("macro:")) {
            val name = id.substring(6)
            val m = AutoClickStore.load(prefsObj()).firstOrNull { it.name == name }
            if (m == null) fail("Function not found") else runMacro(m)
        } else if (id.startsWith("ext:")) {
            val eid = id.substring(4)
            if (eid == "yt_skip") {
                val p = prefsObj()
                p.edit().putBoolean("ext_youtube_skip", !p.getBoolean("ext_youtube_skip", false)).apply()
                playSound(soundActivate)
                speak(extLabel("yt_skip"))
            } else {
                fail("Extension not found")
            }
        }
    }

    private fun moveNext() = moveBy(1)

    private fun movePrevious() = moveBy(-1)

    // Can this node scroll up/down (vertical lists only) in the given direction?
    private fun canScrollVertically(node: AccessibilityNodeInfo, dir: Int): Boolean {
        val ids = node.actionList.map { it.id }
        val down = ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_DOWN.id)
        val up = ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_UP.id)
        val left = ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_LEFT.id)
        val right = ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_RIGHT.id)
        if (down || up) return if (dir > 0) down else up
        if (left || right) return false // horizontal only (chips, carousels)
        return ids.contains(if (dir > 0) AccessibilityNodeInfo.ACTION_SCROLL_FORWARD else AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD)
    }

    // The outermost vertical list around this item that can still scroll in this direction.
    private fun findVerticalScroller(from: AccessibilityNodeInfo?, dir: Int): AccessibilityNodeInfo? {
        var cur: AccessibilityNodeInfo? = from
        var best: AccessibilityNodeInfo? = null
        var d = 0
        while (cur != null && d < 30) {
            if (cur.isScrollable && canScrollVertically(cur, dir)) best = cur
            cur = cur.parent
            d++
        }
        return best
    }

    private fun scrollList(scroller: AccessibilityNodeInfo, dir: Int): Boolean {
        val ids = scroller.actionList.map { it.id }
        val directional = if (dir > 0) AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_DOWN.id
        else AccessibilityNodeInfo.AccessibilityAction.ACTION_SCROLL_UP.id
        if (ids.contains(directional) && scroller.performAction(directional)) return true
        return scroller.performAction(
            if (dir > 0) AccessibilityNodeInfo.ACTION_SCROLL_FORWARD else AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD
        )
    }

    private fun moveBy(dir: Int) {
        if (busyScrolling) return
        syncList()
        if (nodeList.isEmpty()) {
            playSound(soundAlert)
            speak("No items found on this screen")
            return
        }
        val from = focusedNode ?: nodeList.getOrNull(currentIndex)
        val target = currentIndex + dir
        val nextNode = nodeList.getOrNull(target)
        // If the item after this one is not inside the same vertical list (or there is none)
        // and that list can still scroll, scroll it to bring in more items first.
        val scroller = findVerticalScroller(from, dir)
        val needScroll = scroller != null && (nextNode == null || !isDescendantOf(nextNode, scroller))
        if (!needScroll) {
            if (nextNode != null) {
                currentIndex = target
                announceCurrent()
            } else {
                wrapAround(dir)
            }
            return
        }
        if (scroller != null && scrollList(scroller, dir)) {
            busyScrolling = true
            handler.postDelayed({
                busyScrolling = false
                val found = syncList()
                val next = currentIndex + dir
                if (found && next in nodeList.indices && isDescendantOf(nodeList[next], scroller)) {
                    currentIndex = next
                    announceCurrent()
                } else {
                    // The old item scrolled out of view (or nothing new came): pick the
                    // first (or last) item inside the scrolled list.
                    val inside = nodeList.indices.filter { isDescendantOf(nodeList[it], scroller) }
                    val pick = if (dir > 0) inside.firstOrNull() else inside.lastOrNull()
                    if (pick != null) {
                        currentIndex = pick
                        announceCurrent()
                    } else if (found && next in nodeList.indices) {
                        currentIndex = next
                        announceCurrent()
                    } else {
                        fail(if (dir > 0) "End of list" else "Start of list")
                    }
                }
            }, 700)
            return
        }
        if (nextNode != null) {
            currentIndex = target
            announceCurrent()
        } else {
            wrapAround(dir)
        }
    }

    private fun wrapAround(dir: Int) {
        if (nodeList.isEmpty()) return
        currentIndex = if (dir > 0) 0 else nodeList.size - 1
        announceCurrent()
    }

    // What is spoken for an item: its label, and for sliders also the value and how to change it.
    private fun spokenLabel(node: AccessibilityNodeInfo): String {
        val label = labelOf(node)
        if (isSlider(node)) {
            val value = sliderValueOrNull(node) ?: "value unknown"
            return (label ?: "Slider") + ", slider, " + value + ", swipe up to increase, swipe down to decrease"
        }
        return label ?: "Item"
    }

    private fun announceCurrent() {
        val node = nodeList.getOrNull(currentIndex) ?: return
        focusedNode = node
        resetFocusState()
        node.performAction(AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS)
        playSound(soundFocus)
        speak(spokenLabel(node))
    }

    // ======================= Sliders =======================

    private fun isSlider(node: AccessibilityNodeInfo): Boolean {
        val cls = node.className?.toString() ?: ""
        if (cls.contains("SeekBar") || cls.contains("Slider")) return true
        val ids = node.actionList.map { it.id }
        if (ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SET_PROGRESS.id)) return true
        val ri = node.rangeInfo
        if (ri != null && (ids.contains(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD) ||
                ids.contains(AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD))) return true
        return false
    }

    private fun formatMillis(ms: Float): String {
        val total = Math.round(ms / 1000f)
        val h = total / 3600
        val m = (total % 3600) / 60
        val sec = total % 60
        val parts = mutableListOf<String>()
        if (h > 0) parts.add("" + h + (if (h == 1) " hour" else " hours"))
        if (m > 0) parts.add("" + m + (if (m == 1) " minute" else " minutes"))
        if (sec > 0 || parts.isEmpty()) parts.add("" + sec + (if (sec == 1) " second" else " seconds"))
        return parts.joinToString(" ")
    }

    private val timeRegex = Regex("^\\d{1,2}(:\\d{2}){1,2}$")

    private fun collectTimeTexts(node: AccessibilityNodeInfo?, out: MutableList<String>, depth: Int) {
        if (node == null || depth > 4 || out.size >= 2) return
        val t = ownLabel(node)
        if (t != null && timeRegex.matches(t.trim())) out.add(t.trim())
        for (i in 0 until node.childCount) collectTimeTexts(node.getChild(i), out, depth + 1)
    }

    // Players such as YouTube show "0:05" and "10:00" as separate text next to the seek bar.
    private fun nearbyTimeText(node: AccessibilityNodeInfo): String? {
        var cur: AccessibilityNodeInfo? = node.parent
        var level = 0
        while (cur != null && level < 3) {
            val found = mutableListOf<String>()
            collectTimeTexts(cur, found, 0)
            if (found.isNotEmpty()) {
                return if (found.size >= 2) found[0] + " of " + found[1] else found[0]
            }
            cur = cur.parent
            level++
        }
        return null
    }

    // The current value of a slider as text, or null if the app exposes nothing readable.
    private fun sliderValueOrNull(node: AccessibilityNodeInfo): String? {
        if (Build.VERSION.SDK_INT >= 30) {
            val sd = node.stateDescription?.toString()
            if (!sd.isNullOrBlank()) return sd
        }
        val ri = node.rangeInfo
        if (ri != null && ri.max > ri.min) {
            val pct = Math.round((ri.current - ri.min) / (ri.max - ri.min) * 100f)
            val range = ri.max - ri.min
            return if (range >= 10000f) {
                "" + pct + " percent, " + formatMillis(ri.current) + " of " + formatMillis(ri.max)
            } else {
                "" + pct + " percent"
            }
        }
        nearbyTimeText(node)?.let { return it }
        return node.text?.toString()?.takeIf { it.isNotBlank() }
    }

    // increase = true raises the value, false lowers it. Returns false only if nothing worked.
    private fun adjustSlider(node: AccessibilityNodeInfo, increase: Boolean): Boolean {
        node.refresh()
        val ri = node.rangeInfo
        val ids = node.actionList.map { it.id }
        val canSet = ids.contains(AccessibilityNodeInfo.AccessibilityAction.ACTION_SET_PROGRESS.id)
        val canScroll = ids.contains(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD) ||
            ids.contains(AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD)
        var ok = false
        if (ri != null && ri.max > ri.min) {
            val range = ri.max - ri.min
            if (increase && ri.current >= ri.max) {
                fail("Maximum")
                return true
            }
            if (!increase && ri.current <= ri.min) {
                fail("Minimum")
                return true
            }
            val preferScroll = range > 1000f && canScroll
            if (canSet && !preferScroll) {
                val isInt = ri.type == AccessibilityNodeInfo.RangeInfo.RANGE_TYPE_INT
                val step = if (isInt && range <= 20f) 1f else range / 20f
                var target = ri.current + (if (increase) step else -step)
                target = target.coerceIn(ri.min, ri.max)
                if (isInt) target = Math.round(target).toFloat()
                val args = Bundle()
                args.putFloat(AccessibilityNodeInfo.ACTION_ARGUMENT_PROGRESS_VALUE, target)
                ok = node.performAction(AccessibilityNodeInfo.AccessibilityAction.ACTION_SET_PROGRESS.id, args)
            }
        }
        if (!ok) {
            val act = if (increase) AccessibilityNodeInfo.ACTION_SCROLL_FORWARD else AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD
            ok = node.performAction(act)
        }
        if (!ok) return false
        playSound(soundFocus)
        sliderSpoken = false
        sliderWatchUntil = System.currentTimeMillis() + 1500
        // Speak the NEW value. Apps update it a moment after the action, so read it
        // after a short delay, and try once more if nothing readable showed up yet.
        handler.postDelayed({
            if (!sliderSpoken) {
                node.refresh()
                val v = sliderValueOrNull(node)
                if (v != null) {
                    sliderSpoken = true
                    speak(v)
                }
            }
        }, 300)
        handler.postDelayed({
            if (!sliderSpoken) {
                node.refresh()
                sliderSpoken = true
                speak(sliderValueOrNull(node) ?: "Value changed")
            }
        }, 900)
        return true
    }

    // Reading control "Slider": works on the focused slider, else the first slider found
    // on screen, else (if nothing is detected) tries the focused item directly.
    private fun sliderControlStep(increase: Boolean) {
        val cur = focusedNode ?: nodeList.getOrNull(currentIndex)
        var target: AccessibilityNodeInfo? = if (cur != null && isSlider(cur)) cur else null
        if (target == null) {
            rebuildNodeList()
            val idx = nodeList.indexOfFirst { isSlider(it) }
            if (idx >= 0) {
                target = nodeList[idx]
                currentIndex = idx
                focusedNode = target
                target.performAction(AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS)
            }
        }
        if (target != null) {
            if (!adjustSlider(target, increase)) fail("Could not change the slider")
            return
        }
        if (cur != null && adjustSlider(cur, increase)) return
        fail("No slider found. Focus a slider first")
    }

    private fun activateCurrent() {
        val node = focusedNode ?: nodeList.getOrNull(currentIndex)
        if (node == null) {
            playSound(soundAlert)
            speak("First swipe to or touch an item")
            return
        }
        if (readingIndex == actionsControlIndex && actionIndex >= 0) {
            val chosen = actionsFor(node).getOrNull(actionIndex)
            if (chosen != null) {
                val r = runEditAction(node, chosen)
                if (r.ok) {
                    playSound(soundActivate)
                    speak(r.msg)
                } else {
                    fail(r.msg)
                }
                return
            }
        }
        val didClick = clickNodeOrAncestor(node)
        if (didClick) {
            playSound(soundActivate)
        } else {
            playSound(soundAlert)
        }
    }

    // ======================= Reading controls =======================

    private fun resetReading() {
        readUnits = emptyList()
        readIndex = -1
        actionIndex = -1
    }

    // Called whenever focus moves to a different item: also ends selection mode.
    private fun resetFocusState() {
        resetReading()
        selecting = false
    }

    private fun changeReadingControl(dir: Int) {
        // Only the reading controls switched on in "Customize reading controls" are used.
        val off = MenuConfig.disabled(getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE), "reading")
        val ids = listOf("normal", "character", "word", "line", "paragraph", "actions", "slider")
        val enabled = ids.indices.filter { it == 0 || !off.contains(ids[it]) }
        val pos = enabled.indexOf(readingIndex).let { if (it < 0) 0 else it }
        readingIndex = enabled[wrap(pos + dir, enabled.size)]
        resetReading()
        playSound(soundFocus)
        speak(readingLabels[readingIndex])
    }

    private fun nodeText(node: AccessibilityNodeInfo): String? {
        node.refresh()
        val t = if (node.isShowingHintText) null else node.text?.toString()
        return t?.takeIf { it.isNotBlank() }
            ?: node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
            ?: mergedText(node)
    }

    private fun readStep(dir: Int) {
        // Swipe up = increase, swipe down = decrease (dir is +1 for down).
        if (readingIndex == sliderControlIndex) {
            sliderControlStep(dir < 0)
            return
        }
        if (readingIndex == 0) {
            val focused = focusedNode ?: nodeList.getOrNull(currentIndex)
            if (focused != null && isSlider(focused)) {
                if (!adjustSlider(focused, dir < 0)) fail("Could not change the slider")
                return
            }
            fail("Normal item. Swipe down then up to choose character, word, line, paragraph, actions or slider")
            return
        }
        val node = focusedNode ?: nodeList.getOrNull(currentIndex)
        if (node == null) {
            fail("Focus an item first")
            return
        }
        if (readingIndex == actionsControlIndex) {
            stepAction(node, dir)
            return
        }
        val kind = SplitKind.values()[readingIndex - 1]
        // Edit boxes: swipes move the real cursor (and extend the selection in selection mode).
        if (node.isEditable) {
            cursorStep(node, kind, dir)
            return
        }
        if (readUnits.isEmpty()) {
            val text = nodeText(node)
            if (text == null) {
                fail("This item has no readable text")
                return
            }
            readUnits = splitText(text, kind, true)
            if (readUnits.isEmpty()) {
                fail("This item has no readable text")
                return
            }
            readIndex = if (dir > 0) -1 else readUnits.size
        }
        val next = readIndex + dir
        if (next < 0) {
            fail("Start")
            return
        }
        if (next >= readUnits.size) {
            fail("End")
            return
        }
        readIndex = next
        playSound(soundFocus)
        speak(speakable(readUnits[next]))
    }

    private fun clickNodeOrAncestor(node: AccessibilityNodeInfo): Boolean {
        node.refresh()
        var cur: AccessibilityNodeInfo? = node
        var depth = 0
        while (cur != null && depth < 6) {
            if (cur.isClickable && cur.performAction(AccessibilityNodeInfo.ACTION_CLICK)) return true
            cur = cur.parent
            depth++
        }
        return node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
    }

    private fun hintModalEntry() {
        val now = System.currentTimeMillis()
        if (now - lastModalHoverHintAt < 2500) return
        lastModalHoverHintAt = now
        val current = when (mode) {
            Mode.MENU -> menuLabels[menuIndex]
            Mode.UNIT_CHOOSER -> unitLabels[unitIndex]
            Mode.ITEM_SELECT -> describeSelectEntry(selectIndex)
            Mode.AI_MENU -> aiLabels[aiIndex]
            Mode.EXT_MENU -> extLabel(extIds.getOrNull(extIndex) ?: "back")
            Mode.VOICE_MENU -> voiceLabels.getOrElse(voiceIndex) { "" }
            Mode.NORMAL -> ""
        }
        speak("Menu is open. Swipe left or right to change, double tap to choose. Now: " + current)
    }

    // ======================= Screen Reader Menu =======================

    private fun wrap(i: Int, n: Int): Int = ((i % n) + n) % n

    private fun openMenu() {
        menuTarget = focusedNode ?: nodeList.getOrNull(currentIndex)
        refreshMenuIds()
        mode = Mode.MENU
        menuIndex = 0
        playSound(soundScreenChange)
        speak("Screen reader menu. Swipe left or right to change options, double tap to choose. " + menuLabels[0])
    }

    private fun closeMenu() {
        mode = Mode.NORMAL
        menuTarget = null
        playSound(soundScreenChange)
        speak("Menu closed")
    }

    private fun goBack() {
        when (mode) {
            Mode.ITEM_SELECT -> {
                mode = Mode.UNIT_CHOOSER
                playSound(soundFocus)
                speak("Choose unit. " + unitLabels[unitIndex])
            }
            Mode.UNIT_CHOOSER -> {
                mode = Mode.MENU
                playSound(soundFocus)
                speak("Menu. " + menuLabels[menuIndex])
            }
            Mode.VOICE_MENU -> {
                mode = Mode.AI_MENU
                playSound(soundFocus)
                speak("AI menu. " + aiLabels[aiIndex])
            }
            Mode.AI_MENU, Mode.EXT_MENU -> {
                mode = Mode.MENU
                playSound(soundFocus)
                speak("Menu. " + menuLabels[menuIndex])
            }
            Mode.MENU -> closeMenu()
            Mode.NORMAL -> {}
        }
    }

    private fun stepModal(dir: Int) {
        playSound(soundFocus)
        when (mode) {
            Mode.MENU -> {
                menuIndex = wrap(menuIndex + dir, menuLabels.size)
                speak(menuLabels[menuIndex])
            }
            Mode.UNIT_CHOOSER -> {
                unitIndex = wrap(unitIndex + dir, unitLabels.size)
                speak(unitLabels[unitIndex])
            }
            Mode.ITEM_SELECT -> {
                selectIndex = wrap(selectIndex + dir, selectUnits.size + 2)
                speak(describeSelectEntry(selectIndex))
            }
            Mode.AI_MENU -> {
                aiIndex = wrap(aiIndex + dir, aiLabels.size)
                speak(aiLabels[aiIndex])
            }
            Mode.EXT_MENU -> {
                extIndex = wrap(extIndex + dir, extIds.size)
                speak(extLabel(extIds[extIndex]))
            }
            Mode.VOICE_MENU -> {
                voiceIndex = wrap(voiceIndex + dir, voiceLabels.size)
                speak(voiceLabels[voiceIndex])
            }
            Mode.NORMAL -> {}
        }
    }

    private fun activateModal() {
        when (mode) {
            Mode.MENU -> when (menuIds.getOrNull(menuIndex)) {
                "copy" -> doCopy()
                "append" -> doAppend()
                "cut" -> doCut()
                "paste" -> doPaste()
                "select" -> startSelect()
                "clear" -> doClear()
                "ai" -> openAiMenu()
                "extensions" -> openExtMenu()
                else -> closeMenu()
            }
            Mode.EXT_MENU -> activateExt()
            Mode.VOICE_MENU -> activateVoiceMenu()
            Mode.AI_MENU -> when (aiIndex) {
                0 -> runAi(false)
                1 -> runAi(true)
                2 -> beginVoiceFlow()
                else -> goBack()
            }
            Mode.UNIT_CHOOSER -> beginItemSelect(SplitKind.values()[unitIndex])
            Mode.ITEM_SELECT -> {
                val n = selectUnits.size
                when {
                    selectIndex < n -> toggleSelected(selectIndex)
                    selectIndex == n -> copySelection(false)
                    else -> copySelection(true)
                }
            }
            Mode.NORMAL -> {}
        }
    }

    private fun fail(message: String) {
        playSound(soundAlert)
        speak(message)
    }

    private fun finish(message: String) {
        mode = Mode.NORMAL
        menuTarget = null
        playSound(soundActivate)
        speak(message)
    }

    private fun targetText(): String? {
        val node = menuTarget
        if (node == null) {
            fail("Focus an item first, then open the menu")
            return null
        }
        node.refresh()
        val t = if (node.isShowingHintText) null else node.text?.toString()
        val s = t?.takeIf { it.isNotBlank() }
            ?: node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
            ?: mergedText(node)
        if (s == null) {
            fail("This item has no text that can be copied")
            return null
        }
        return s
    }

    private fun writeClipboard(): Boolean {
        return try {
            val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
            cm.setPrimaryClip(ClipData.newPlainText("screen_reader", clipList.joinToString("\n")))
            suppressWindowEventsUntil = System.currentTimeMillis() + 2000
            true
        } catch (e: Exception) {
            false
        }
    }

    // ======================= Auto focus on a new screen =======================

    private fun scheduleAutoFocus(title: String) {
        autoFocusPending?.let { handler.removeCallbacks(it) }
        val r = Runnable { autoFocus(title) }
        autoFocusPending = r
        handler.postDelayed(r, 350)
    }

    // Instead of announcing "new screen", put focus somewhere sensible and read it.
    private fun autoFocus(title: String) {
        if (mode != Mode.NORMAL || gamingActive) return
        val now = System.currentTimeMillis()
        if (now - lastAutoFocusAt < 1200) return
        lastAutoFocusAt = now
        rebuildNodeList()
        currentIndex = -1
        focusedNode = null
        resetFocusState()
        playSound(soundScreenChange)
        if (nodeList.isEmpty()) {
            if (title.isNotEmpty()) speak(title)
            return
        }
        var idx = 0
        try {
            val input = rootInActiveWindow?.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
            if (input != null) {
                val i = nodeList.indexOfFirst { it == input }
                if (i >= 0) idx = i
            }
        } catch (e: Exception) {
            // keep first item
        }
        currentIndex = idx
        val node = nodeList[idx]
        focusedNode = node
        node.performAction(AccessibilityNodeInfo.ACTION_ACCESSIBILITY_FOCUS)
        val label = spokenLabel(node)
        speak(if (title.isNotEmpty() && title != label) "$title. $label" else label)
    }

    // ======================= Typing feedback =======================

    private fun typedCharText(c: Char): String = when {
        c.isUpperCase() -> "capital $c"
        c == ' ' -> "space"
        c == '\n' -> "new line"
        else -> c.toString()
    }

    private fun echoTyping(event: AccessibilityEvent) {
        if (voiceActive) return
        if (event.isPassword) return
        val now = event.text.firstOrNull()?.toString() ?: return
        val from = event.fromIndex
        val added = event.addedCount
        val removed = event.removedCount
        if (added > 0 && from >= 0 && from + added <= now.length) {
            val ins = now.substring(from, from + added)
            if (ins.length == 1) {
                val c = ins[0]
                if (c == ' ' || c == '\n') {
                    // Space or Enter: say the word that was just finished.
                    val before = now.substring(0, from).trimEnd()
                    val word = before.split(' ', '\n').lastOrNull() ?: ""
                    speak(if (word.isNotBlank()) word else typedCharText(c))
                } else {
                    speak(typedCharText(c))
                }
            } else if (ins.isNotBlank()) {
                speak(if (ins.length > 300) ins.substring(0, 300) else ins.trim())
            }
        } else if (removed > 0 && added == 0) {
            val before = event.beforeText?.toString()
            if (before != null && from >= 0 && from < before.length) {
                val gone = before.substring(from, minOf(before.length, from + removed))
                speak(if (gone.length == 1) "deleted " + typedCharText(gone[0]) else "deleted")
            } else {
                speak("deleted")
            }
        }
    }

    // ======================= Gaming mode =======================

    private fun isGamePackage(pkg: String): Boolean {
        if (pkg == packageName) return false
        val prefs = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
        val on = prefs.getStringSet("gaming_on", emptySet()) ?: emptySet()
        val off = prefs.getStringSet("gaming_off", emptySet()) ?: emptySet()
        if (off.contains(pkg)) return false
        if (on.contains(pkg)) return true
        return try {
            packageManager.getApplicationInfo(pkg, 0).category == ApplicationInfo.CATEGORY_GAME
        } catch (e: Exception) {
            false
        }
    }

    private fun updateGamingMode(pkg: String?) {
        if (pkg == null || pkg == packageName || pkg == "com.android.systemui") return
        if (pkg == lastForegroundPackage) return
        lastForegroundPackage = pkg
        if (gamingSuppressedPackage != null && gamingSuppressedPackage != pkg) gamingSuppressedPackage = null
        val want = gamingSuppressedPackage != pkg && isGamePackage(pkg)
        if (want && !gamingActive) enterGaming() else if (!want && gamingActive) exitGaming()
    }

    private fun setTouchExploration(on: Boolean) {
        try {
            val info = serviceInfo ?: return
            val f = AccessibilityServiceInfo.FLAG_REQUEST_TOUCH_EXPLORATION_MODE or
                AccessibilityServiceInfo.FLAG_SERVICE_HANDLES_DOUBLE_TAP
            info.flags = if (on) info.flags or f else info.flags and f.inv()
            serviceInfo = info
        } catch (e: Exception) {
            // ignore
        }
    }

    private fun enterGaming() {
        gamingActive = true
        mode = Mode.NORMAL
        menuTarget = null
        autoFocusPending?.let { handler.removeCallbacks(it) }
        setTouchExploration(false)
        playSound(soundScreenChange)
        speak("Gaming mode on. Screen reader touch gestures are off. Press both volume keys together to turn it off.")
    }

        gamingActive = false

    }

    // In gaming mode there is no touch exploration, so read text the game itself puts on screen.
    private fun speakGameText(event: AccessibilityEvent) {
        val types = event.contentChangeTypes
        val textual = (types and AccessibilityEvent.CONTENT_CHANGE_TYPE_TEXT) != 0 ||
            (types and AccessibilityEvent.CONTENT_CHANGE_TYPE_CONTENT_DESCRIPTION) != 0
        if (!textual) return
        var t = event.text.joinToString(" ").trim()
        if (t.isEmpty()) t = event.contentDescription?.toString()?.trim() ?: ""
        if (t.isEmpty()) return
        val now = System.currentTimeMillis()
        if (t == lastGameText && now - lastGameTextAt < 1000) return
        lastGameText = t
        lastGameTextAt = now
        speak(t)
    }

    // ---- YouTube: press "Skip ad" automatically ----

    private val skipLoop = object : Runnable {
        override fun run() {
            try {
                if (!gamingActive && lastForegroundPackage == youtubePackage &&
                    prefsObj().getBoolean("ext_youtube_skip", false)
                ) {
                    trySkipAd()
                }
            } catch (e: Exception) {
                // never let the loop die
            }
            handler.postDelayed(this, 1000)
        }
    }

    private fun nfc(s: String): String = Normalizer.normalize(s, Normalizer.Form.NFC).lowercase()

    private fun isSkipAdLabel(raw: String): Boolean {
        val l = nfc(raw).trim()
        if (l == "skip" || l == "skip ad" || l == "skip ads" || l.startsWith("skip ad")) return true
        return l.contains(nfc("\u0935\u093f\u091c\u094d\u091e\u093e\u092a\u0928")) && l.contains(nfc("\u091b\u094b\u0921\u093c"))
    }

    private fun trySkipAd() {
        val root = rootInActiveWindow ?: return
        if (root.packageName?.toString() != youtubePackage) return
        val candidates = mutableListOf<AccessibilityNodeInfo>()
        candidates.addAll(root.findAccessibilityNodeInfosByText("skip"))
        candidates.addAll(root.findAccessibilityNodeInfosByText("\u0935\u093f\u091c\u094d\u091e\u093e\u092a\u0928"))
        for (c in candidates) {
            if (!c.isVisibleToUser) continue
            val labels = listOfNotNull(ownLabel(c), c.contentDescription?.toString())
            if (labels.any { isSkipAdLabel(it) }) {
                if (clickNodeOrAncestor(c)) {
                    playSound(soundActivate)
                    speak("Ad skipped")
                    return
                }
            }
        }
    }

    // ---- Auto click functions ----

    private fun normLabel(s: String): String = s.trim().replace(Regex("\\s+"), " ").lowercase()

    private fun labelsOfNode(n: AccessibilityNodeInfo): List<String> {
        val l = mutableListOf<String>()
        ownLabel(n)?.let { l.add(it) }
        n.contentDescription?.toString()?.takeIf { it.isNotBlank() }?.let { l.add(it) }
        labelOf(n)?.let { l.add(it) }
        mergedText(n)?.let { l.add(it) }
        return l.map { normLabel(it) }
    }

    // Finds the item copied from the screen: an exact match first, then the shortest label containing it.
    private fun findStepNode(step: String, avoid: AccessibilityNodeInfo?, allowAvoid: Boolean): AccessibilityNodeInfo? {
        rebuildNodeList()
        val want = normLabel(step)
        var partial: AccessibilityNodeInfo? = null
        var partialLen = Int.MAX_VALUE
        for (n in nodeList) {
            if (avoid != null && n == avoid && !allowAvoid) continue
            val labels = labelsOfNode(n)
            if (labels.any { it == want }) return n
            val shortest = labels.filter { it.contains(want) }.minOfOrNull { it.length }
            if (shortest != null && shortest < partialLen) {
                partial = n
                partialLen = shortest
            }
        }
        return partial
    }

    private fun runMacro(m: Macro) {
        if (m.steps.isEmpty()) {
            fail("This function has no steps")
            return
        }
        if (macroRunning) {
            fail("A function is already running")
            return
        }
        macroRunning = true
        mode = Mode.NORMAL
        menuTarget = null
        playSound(soundActivate)
        speak("Running " + m.name)
        handler.postDelayed({ macroStep(m.steps, 0, null, System.currentTimeMillis()) }, 1200)
    }

    private fun macroStep(steps: List<String>, idx: Int, last: AccessibilityNodeInfo?, since: Long) {
        if (idx >= steps.size) {
            macroRunning = false
            playSound(soundActivate)
            speak("Function finished")
            return
        }
        val elapsed = System.currentTimeMillis() - since
        // If only the item we just clicked matches, wait a little in case the screen is still changing.
        val node = findStepNode(steps[idx], last, elapsed > 1500)
        if (node != null) {
            clickNodeOrAncestor(node)
            playSound(soundFocus)
            handler.postDelayed({ macroStep(steps, idx + 1, node, System.currentTimeMillis()) }, 900)
            return
        }
        if (elapsed > 5000) {
            macroRunning = false
            fail("Could not find " + steps[idx] + " on the screen. Stopped at step " + (idx + 1))
            return
        }
        handler.postDelayed({ macroStep(steps, idx, last, since) }, 300)
    }

    private fun startVoiceTyping() {

            fail("No API key found. Open the screen reader app and choose Add API key first.")

        if (!isKeyboardOpen()) {
            fail("Keyboard not found. Open the keyboard on the screen first, then start voice typing.")
            return

        if (findInputField() == null) {

            return

        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) {

            } catch (e: Exception) {
                fail("Could not open the permission screen.")
            }
            return

        val lang = prefsObj().getString("voice_lang_name", "Hindi") ?: "Hindi"
        voiceActive = true
        voiceStopRequested = false
        voiceCaptureDone = false
        voicePending.set(0)
        voiceTyped.setLength(0)
        voiceLastError = null
        speak("Voice typing started. Speak in " + lang + ". It stops after several seconds of silence, or on any touch.")
        // Start listening only after the announcement has finished, so the microphone does not hear it.
        afterSpeech {
            if (!voiceActive || voiceStopRequested) {
                voiceCaptureDone = true
                maybeFinishVoice()
            } else {
                playSound(soundFocus)
                Thread { captureLoop(lang) }.start()
            }
        }
    }

    private fun afterSpeech(tries: Int = 0, action: () -> Unit) {
        handler.postDelayed({
            val speaking = try {
                tts?.isSpeaking == true
            } catch (e: Exception) {
                false
            }
            if (speaking && tries < 60) afterSpeech(tries + 1, action) else handler.postDelayed({ action() }, 300)
        }, if (tries == 0) 400L else 150L)
    }

    private fun endCapture(error: String?) {

    private fun maybeFinishVoice() {
        if (!voiceActive || !voiceCaptureDone || voicePending.get() != 0) return
        voiceActive = false
        playSound(soundActivate)
        val err = voiceLastError
        val typed = voiceTyped.toString().trim()
        if (err != null) {
            speak("Voice typing stopped. " + err)
        } else if (typed.isNotEmpty()) {
            speak("Voice typing stopped. Typed: " + (if (typed.length > 250) typed.substring(0, 250) else typed))
        } else {
            speak("Voice typing stopped. Nothing was typed.")
        }
    }

    // Records from the microphone, cuts the speech into pieces at short pauses, sends each piece
    // for transcription, and stops on its own after about six seconds of silence.
    private fun captureLoop(langName: String) {
        val rate = 16000
        var rec: AudioRecord? = null
        try {
            val minBuf = AudioRecord.getMinBufferSize(rate, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT)
            rec = AudioRecord(
                MediaRecorder.AudioSource.MIC, rate, AudioFormat.CHANNEL_IN_MONO,
                AudioFormat.ENCODING_PCM_16BIT, maxOf(minBuf, rate)
            )
            if (rec.state != AudioRecord.STATE_INITIALIZED) {
                endCapture("Could not start the microphone.")
                return
            }
            rec.startRecording()
            val chunk = ShortArray(rate / 10)
            var seg = ByteArrayOutputStream()
            val preRoll = LinkedList<ByteArray>()
            var haveSpeech = false
            var speechMs = 0
            var segMs = 0
            var silenceMs = 0
            var idleMs = 0
            var noiseMin = Double.MAX_VALUE
            var calibrated = 0
            var threshold = 700.0

            fun flush() {
                if (speechMs >= 300) submitSegment(seg.toByteArray(), langName)
                seg = ByteArrayOutputStream()
                haveSpeech = false
                speechMs = 0
                segMs = 0
                silenceMs = 0
            }

            while (!voiceStopRequested) {
                val n = rec.read(chunk, 0, chunk.size)
                if (n < 0) break
                if (n == 0) continue
                var sum = 0.0
                val bytes = ByteArray(n * 2)
                for (i in 0 until n) {
                    val v = chunk[i].toInt()
                    sum += (v * v).toDouble()
                    bytes[2 * i] = (v and 0xFF).toByte()
                    bytes[2 * i + 1] = ((v shr 8) and 0xFF).toByte()
                }
                val rms = Math.sqrt(sum / n)
                if (calibrated < 8) {
                    if (rms < noiseMin) noiseMin = rms
                    calibrated++
                    if (calibrated == 8) threshold = maxOf(500.0, noiseMin * 3.0)
                }
                val isSpeech = calibrated >= 8 && rms > threshold
                if (isSpeech) {
                    if (!haveSpeech) {
                        haveSpeech = true
                        for (p in preRoll) seg.write(p)
                        preRoll.clear()
                    }
                    seg.write(bytes)
                    segMs += 100
                    speechMs += 100
                    silenceMs = 0
                    idleMs = 0
                } else {
                    if (haveSpeech) {
                        seg.write(bytes)
                        segMs += 100
                        silenceMs += 100
                    } else {
                        preRoll.addLast(bytes)
                        if (preRoll.size > 3) preRoll.removeFirst()
                    }
                    idleMs += 100
                }
                if (haveSpeech && (silenceMs >= 1200 || segMs >= 25000)) flush()
                if (idleMs >= 6000) break
            }
            if (haveSpeech) flush()
            endCapture(null)
        } catch (e: SecurityException) {
            endCapture("Microphone permission is missing.")
        } catch (e: Exception) {
            endCapture("The microphone stopped working.")
        } finally {
            try {
                rec?.stop()
            } catch (e: Exception) {
                // ignore
            }
            try {
                rec?.release()
            } catch (e: Exception) {
                // ignore
            }
        }
    }

    private fun wavFrom(pcm: ByteArray, rate: Int): ByteArray {
        val out = ByteArrayOutputStream()
        fun i32(v: Int) {
            out.write(v and 0xFF)
            out.write((v shr 8) and 0xFF)
            out.write((v shr 16) and 0xFF)
            out.write((v shr 24) and 0xFF)
        }
        fun i16(v: Int) {
            out.write(v and 0xFF)
            out.write((v shr 8) and 0xFF)
        }
        out.write("RIFF".toByteArray())
        i32(36 + pcm.size)
        out.write("WAVE".toByteArray())
        out.write("fmt ".toByteArray())
        i32(16)
        i16(1)
        i16(1)
        i32(rate)
        i32(rate * 2)
        i16(2)
        i16(16)
        out.write("data".toByteArray())
        i32(pcm.size)
        out.write(pcm)
        return out.toByteArray()
    }

    private fun voicePrompt(lang: String): String =
        "Transcribe this speech into text that will be typed into a text box. The speaker's main language is " + lang +
            ": write it in its normal script. When the speaker says English words or phrases, write them in English " +
            "using Latin letters exactly as spoken, and do not translate them. Add natural punctuation. " +
            "Output only the transcribed text, with no quotes, labels or explanations. " +
            "If there is no clear speech, output nothing."

    private fun submitSegment(pcm: ByteArray, langName: String) {
        val key = prefsObj().getString("gemini_api_key", null)?.trim() ?: return
        voicePending.incrementAndGet()
        voiceExecutor.execute {
            val b64 = Base64.encodeToString(wavFrom(pcm, 16000), Base64.NO_WRAP)
            val res = callGemini(key, voicePrompt(langName), b64, "audio/wav", false)
            handler.post {
                voicePending.decrementAndGet()
                if (res.ok) {
                    if (res.text.isNotBlank()) insertVoiceText(res.text.trim())
                } else {
                    if (voiceLastError == null) voiceLastError = res.text
                    voiceStopRequested = true
                }
                maybeFinishVoice()
            }
        }
    }

    // Puts the recognised text into the focused text box at the cursor.
    private fun insertVoiceText(text: String) {
        val node = findInputField()
        if (node == null) {
            if (voiceLastError == null) voiceLastError = "The text box lost focus."
            voiceStopRequested = true
            return
        }
        node.refresh()
        var current = node.text?.toString() ?: ""
        if (node.isShowingHintText) current = ""
        var start = node.textSelectionStart
        var end = node.textSelectionEnd
        if (start < 0 || end < 0 || start > current.length || end > current.length) {
            start = current.length
            end = current.length
        }
        if (start > end) {
            val t = start
            start = end
            end = t
        }
        val before = current.substring(0, start)
        val after = current.substring(end)
        val insertion = if (before.isNotEmpty() && !before.last().isWhitespace()) " " + text else text
        val args = Bundle()
        args.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, before + insertion + after)
        var ok = node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
        if (ok) {
            val pos = before.length + insertion.length
            val sel = Bundle()
            sel.putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_START_INT, pos)
            sel.putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_END_INT, pos)
            node.performAction(AccessibilityNodeInfo.ACTION_SET_SELECTION, sel)
        } else {
            // Some apps refuse direct text setting: paste through the clipboard instead.
            try {
                suppressWindowEventsUntil = System.currentTimeMillis() + 2000
                val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
                cm.setPrimaryClip(ClipData.newPlainText("voice", insertion))
                ok = node.performAction(AccessibilityNodeInfo.ACTION_PASTE)
            } catch (e: Exception) {
                ok = false
            }
        }
        if (ok) {
            voiceTyped.append(insertion)
        } else {
            if (voiceLastError == null) voiceLastError = "This text box does not accept typed text."
            voiceStopRequested = true
        }
    }

    private fun runAi(imageOnly: Boolean) {
        val prefs = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
        val key = prefs.getString("gemini_api_key", null)?.trim()
        if (key.isNullOrEmpty()) {
            fail("No API key found. Open the screen reader app and choose Add API key first.")
            return
        }
        if (aiBusy) {
            fail("Already working on a request. Please wait.")
            return
        }
        if (Build.VERSION.SDK_INT < 30) {
            fail("AI descriptions need Android 11 or newer.")
            return
        }
        var crop: Rect? = null
        if (imageOnly) {
            val node = menuTarget
            if (node == null) {
                fail("Focus an image first, then open the menu.")
                return
            }
            node.refresh()
            val r = boundsOf(node)
            if (r.width() < 20 || r.height() < 20) {
                fail("Could not find where this item is on the screen.")
                return
            }
            crop = r
        }
        mode = Mode.NORMAL
        menuTarget = null
        aiBusy = true
        playSound(soundActivate)
        speak(if (imageOnly) "Describing image. Please wait." else "Describing screen. Please wait.")
        val cropFinal = crop
        // Small delay so the spoken hint and menu state settle before the screenshot.
        handler.postDelayed({ captureAndDescribe(key, imageOnly, cropFinal) }, 400)
    }

    private fun aiFail(message: String) {
        handler.post {
            aiBusy = false
            fail(message)
        }
    }

    private fun captureAndDescribe(key: String, imageOnly: Boolean, crop: Rect?) {
        if (Build.VERSION.SDK_INT < 30) {
            aiBusy = false
            return
        }
        try {
            takeScreenshot(Display.DEFAULT_DISPLAY, aiExecutor, object : AccessibilityService.TakeScreenshotCallback {
                override fun onSuccess(result: AccessibilityService.ScreenshotResult) {
                    try {
                        val hw = result.hardwareBuffer
                        val hwBmp = Bitmap.wrapHardwareBuffer(hw, result.colorSpace)
                        val soft = hwBmp?.copy(Bitmap.Config.ARGB_8888, false)
                        hw.close()
                        if (soft == null) {
                            aiFail("Could not read the screenshot.")
                            return
                        }
                        var bmp: Bitmap = soft
                        if (crop != null) {
                            val l = crop.left.coerceIn(0, bmp.width - 1)
                            val t = crop.top.coerceIn(0, bmp.height - 1)
                            val r = crop.right.coerceIn(l + 1, bmp.width)
                            val b = crop.bottom.coerceIn(t + 1, bmp.height)
                            if (r - l < 20 || b - t < 20) {
                                aiFail("This item is not visible on the screen.")
                                return
                            }
                            bmp = Bitmap.createBitmap(bmp, l, t, r - l, b - t)
                        }
                        val maxDim = 1280
                        val longest = maxOf(bmp.width, bmp.height)
                        if (longest > maxDim) {
                            val scale = maxDim.toFloat() / longest
                            bmp = Bitmap.createScaledBitmap(
                                bmp, maxOf(1, (bmp.width * scale).toInt()), maxOf(1, (bmp.height * scale).toInt()), true
                            )
                        }
                        val bos = ByteArrayOutputStream()
                        bmp.compress(Bitmap.CompressFormat.JPEG, 85, bos)
                        val b64 = Base64.encodeToString(bos.toByteArray(), Base64.NO_WRAP)
                        val prompt = if (imageOnly) AI_PROMPT_IMAGE else AI_PROMPT_SCREEN
                        val res = callGemini(key, prompt, b64)
                        handler.post {
                            aiBusy = false
                            if (res.ok) {
                                playSound(soundActivate)
                                speakLong(res.text)
                            } else {
                                fail(res.text)
                            }
                        }
                    } catch (e: Exception) {
                        aiFail("Something went wrong while preparing the picture.")
                    }
                }

                override fun onFailure(errorCode: Int) {
                    aiFail("Could not take a screenshot. Some screens block screenshots. Error " + errorCode)
                }
            })
        } catch (e: Exception) {
            aiBusy = false
            fail("Could not take a screenshot.")
        }
    }

    private val AI_PROMPT_SCREEN =
        "You are helping a blind person who uses a screen reader on an Android phone. " +
            "Describe this screenshot clearly, from top to bottom: which app or screen it is, the main content, " +
            "all important visible text (read it out), the buttons and controls, and briefly what any pictures show. " +
            "Write plain sentences that sound natural when read aloud. Do not use markdown, bullet points, or asterisks."

    private val AI_PROMPT_IMAGE =
        "You are helping a blind person. Describe this image in detail: the main subject, people " +
            "(appearance, expressions, actions), objects, the setting, colors, and transcribe any text visible in it. " +
            "Write plain sentences that sound natural when read aloud. Do not use markdown, bullet points, or asterisks."

    // One try with one model. next = true means "this model cannot help right now, try another".
    private class Attempt(val ok: Boolean, val text: String, val next: Boolean, val busy: Boolean)

    private fun tryModel(model: String, key: String, prompt: String, imageB64: String, mime: String, describeMode: Boolean): Attempt {
        try {
            val url = URL("https://generativelanguage.googleapis.com/v1beta/models/" + model + ":generateContent")
            val conn = url.openConnection() as HttpURLConnection
            conn.requestMethod = "POST"
            conn.connectTimeout = 20000
            conn.readTimeout = 60000
            conn.doOutput = true
            conn.setRequestProperty("Content-Type", "application/json")
            conn.setRequestProperty("x-goog-api-key", key)

            val inline = JSONObject().put("mime_type", mime).put("data", imageB64)
            val parts = JSONArray()
                .put(JSONObject().put("text", prompt))
                .put(JSONObject().put("inline_data", inline))
            val body = JSONObject().put("contents", JSONArray().put(JSONObject().put("parts", parts)))
            conn.outputStream.use { it.write(body.toString().toByteArray(Charsets.UTF_8)) }

            val code = conn.responseCode
            val stream = if (code in 200..299) conn.inputStream else conn.errorStream
            val resp = stream?.bufferedReader()?.use { it.readText() } ?: ""
            conn.disconnect()

            if (code in 200..299) {
                val out = StringBuilder()
                val cands = JSONObject(resp).optJSONArray("candidates")
                val partsOut = cands?.optJSONObject(0)?.optJSONObject("content")?.optJSONArray("parts")
                if (partsOut != null) {
                    for (i in 0 until partsOut.length()) {
                        out.append(partsOut.optJSONObject(i)?.optString("text", "") ?: "")
                    }
                }
                val clean = if (describeMode) out.toString().replace("*", "").replace("#", "").trim() else out.toString().trim()
                if (clean.isNotEmpty() || !describeMode) return Attempt(true, clean, false, false)
                return Attempt(false, "The AI did not return a description. The content may have been blocked.", true, false)
            }
            val apiMsg = try {
                JSONObject(resp).getJSONObject("error").getString("message")
            } catch (e: Exception) {
                ""
            }
            return when (code) {
                400 -> if (apiMsg.contains("API key", true)) {
                    Attempt(false, "Your API key is not valid. Please check it in the app.", false, false)
                } else {
                    Attempt(false, "The request was rejected. " + apiMsg, true, false)
                }
                401, 403 -> Attempt(false, "Your API key was rejected. Please check it in the app.", false, false)
                404 -> Attempt(false, "AI model " + model + " is not available.", true, false)
                429 -> Attempt(false, "Quota exceeded or too many requests.", true, true)
                500, 502, 503, 504 -> Attempt(false, "The AI service is busy.", true, true)
                else -> Attempt(false, "AI service error " + code + ". " + apiMsg, true, false)
            }
        } catch (e: java.net.UnknownHostException) {
            return Attempt(false, "No internet connection.", false, false)
        } catch (e: java.net.SocketTimeoutException) {
            return Attempt(false, "The AI request timed out.", true, true)
        } catch (e: Exception) {
            return Attempt(false, "Network error. Please try again.", true, false)
        }
    }

    // Tries several Gemini models in turn: if one is overloaded (503), over quota (429) or
    // missing (404), the next one is tried automatically. The model that worked last is
    // tried first next time.
    private fun callGemini(key: String, prompt: String, imageB64: String, mime: String = "image/jpeg", describeMode: Boolean = true): AiResult {
        val prefs = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
        val known = listOf(
            "gemini-2.5-flash",
            "gemini-2.5-flash-lite",
            "gemini-flash-latest",
            "gemini-flash-lite-latest",
            "gemini-2.0-flash",
            "gemini-2.0-flash-lite"
        )
        val lastGood = prefs.getString("gemini_last_model", null)
        val models = if (lastGood != null && known.contains(lastGood)) {
            listOf(lastGood) + known.filter { it != lastGood }
        } else {
            known
        }
        var lastError = "Unknown error."
        var sawBusy = false
        for (round in 0 until 2) {
            for (model in models) {
                val r = tryModel(model, key, prompt, imageB64, mime, describeMode)
                if (r.ok) {
                    prefs.edit().putString("gemini_last_model", model).apply()
                    return AiResult(true, r.text)
                }
                lastError = r.text
                if (r.busy) sawBusy = true
                if (!r.next) return AiResult(false, r.text)
            }
            // Every model failed: if it was because of load, wait a moment and go round once more.
            if (round == 0 && sawBusy) {
                try {
                    Thread.sleep(2500)
                } catch (e: InterruptedException) {
                    break
                }
            } else {
                break
            }
        }
        if (sawBusy) {
            return AiResult(false, "All AI models are busy or over quota right now. Please try again in a minute.")
        }
        return AiResult(false, lastError)
    }

    // Speaks long text in chunks (text-to-speech engines refuse very long strings).
    private fun speakLong(text: String) {
        val engine = tts
        if (engine == null) {
            playSound(soundAlert)
            return
        }
        val chunks = mutableListOf<String>()
        var rest = text
        while (rest.length > 3000) {
            var cut = rest.lastIndexOf(". ", 3000)
            if (cut < 500) cut = rest.lastIndexOf(' ', 3000)
            if (cut < 1) cut = 3000
            chunks.add(rest.substring(0, cut + 1))
            rest = rest.substring(cut + 1)
        }
        if (rest.isNotBlank()) chunks.add(rest)
        chunks.forEachIndexed { i, c ->
            val params = Bundle()
            params.putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, ttsVolume)
            val r = engine.speak(
                c,
                if (i == 0) TextToSpeech.QUEUE_FLUSH else TextToSpeech.QUEUE_ADD,
                params,
                "ai" + System.currentTimeMillis() + "_" + i
            )
            if (r == TextToSpeech.ERROR) playSound(soundAlert)
        }
    }

    // ======================= Shared edit operations =======================

    private class OpResult(val ok: Boolean, val msg: String)

    private enum class EditAction {
        CURSOR_START, CURSOR_END, TOGGLE_SELECT, COPY, APPEND, CUT, PASTE, CLEAR
    }

    private fun menuOp(op: (AccessibilityNodeInfo) -> OpResult) {
        val node = menuTarget
        if (node == null) {
            fail("Focus an item first, then open the menu")
            return
        }
        val r = op(node)
        if (r.ok) finish(r.msg) else fail(r.msg)
    }

    private fun doCopy() = menuOp { opCopy(it, false) }
    private fun doAppend() = menuOp { opCopy(it, true) }
    private fun doCut() = menuOp { opCut(it) }
    private fun doPaste() = menuOp { opPaste(it) }
    private fun doClear() = menuOp { opClear(it) }

    private fun editableText(node: AccessibilityNodeInfo): String {
        return if (node.isShowingHintText) "" else (node.text?.toString() ?: "")
    }

    // Returns the selected range (start, end) of a text field, or null if nothing is selected.
    private fun selectionOf(node: AccessibilityNodeInfo): Pair<Int, Int>? {
        val start = node.textSelectionStart
        val end = node.textSelectionEnd
        if (start < 0 || end < 0) return null
        val len = editableText(node).length
        val s0 = minOf(start, end)
        val e0 = maxOf(start, end)
        if (e0 <= s0 || e0 > len) return null
        return Pair(s0, e0)
    }

    private fun cursorPos(node: AccessibilityNodeInfo, len: Int): Int {
        val e = node.textSelectionEnd
        return if (e in 0..len) e else len
    }

    private fun setNodeText(node: AccessibilityNodeInfo, text: String): Boolean {
        val args = Bundle()
        args.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text)
        return node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
    }

    private fun setSelectionRange(node: AccessibilityNodeInfo, s0: Int, e0: Int): Boolean {
        node.refresh()
        if (node.textSelectionStart == s0 && node.textSelectionEnd == e0) return true
        if (!node.isFocused) node.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
        val args = Bundle()
        args.putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_START_INT, s0)
        args.putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_END_INT, e0)
        return node.performAction(AccessibilityNodeInfo.ACTION_SET_SELECTION, args)
    }

    private fun opCopy(node: AccessibilityNodeInfo, append: Boolean): OpResult {
        node.refresh()
        val raw = if (node.isShowingHintText) null else node.text?.toString()
        val sel = selectionOf(node)
        val text: String? = if (raw != null && sel != null) {
            raw.substring(sel.first, sel.second)
        } else {
            raw?.takeIf { it.isNotBlank() }
                ?: node.contentDescription?.toString()?.takeIf { it.isNotBlank() }
                ?: mergedText(node)
        }
        if (text == null) return OpResult(false, "This item has no text that can be copied")
        if (!append) clipList.clear()
        clipList.add(text)
        if (!writeClipboard()) return OpResult(false, "Could not copy")
        return if (append) {
            OpResult(true, "Added. The clipboard now has " + clipList.size + " items")
        } else {
            OpResult(true, "Copied")
        }
    }

    private fun opCut(node: AccessibilityNodeInfo): OpResult {
        if (!node.isEditable) return OpResult(false, "Cannot cut here, this is not an edit box")
        node.refresh()
        val raw = editableText(node)
        val sel = selectionOf(node) ?: return OpResult(false, "Select some text first")
        clipList.clear()
        clipList.add(raw.substring(sel.first, sel.second))
        if (!writeClipboard()) return OpResult(false, "Could not cut")
        val newText = raw.substring(0, sel.first) + raw.substring(sel.second)
        if (!setNodeText(node, newText)) return OpResult(false, "Could not cut")
        selecting = false
        setSelectionRange(node, sel.first, sel.first)
        return OpResult(true, "Cut")
    }

    private fun opPaste(node: AccessibilityNodeInfo): OpResult {
        if (!node.isEditable) {
            return OpResult(false, "Cannot paste here. Focus an edit box and try again")
        }
        node.refresh()
        suppressWindowEventsUntil = System.currentTimeMillis() + 2000
        node.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
        var ok = node.performAction(AccessibilityNodeInfo.ACTION_PASTE)
        if (!ok && clipList.isNotEmpty()) {
            // Fallback: write the collected text straight into the box.
            ok = setNodeText(node, editableText(node) + clipList.joinToString("\n"))
        }
        selecting = false
        return if (ok) OpResult(true, "Pasted") else OpResult(false, "Could not paste")
    }

    private fun opClear(node: AccessibilityNodeInfo): OpResult {
        if (!node.isEditable) return OpResult(false, "This is not an edit box")
        selecting = false
        return if (setNodeText(node, "")) {
            OpResult(true, "Edit box cleared")
        } else {
            OpResult(false, "Could not clear the edit box")
        }
    }

    // ======================= Actions reading-control =======================

    private fun actionsFor(node: AccessibilityNodeInfo): List<EditAction> {
        return if (node.isEditable) {
            EditAction.values().toList()
        } else {
            listOf(EditAction.COPY, EditAction.APPEND)
        }
    }

    private fun actionLabel(a: EditAction): String = when (a) {
        EditAction.CURSOR_START -> "Move cursor to start"
        EditAction.CURSOR_END -> "Move cursor to end"
        EditAction.TOGGLE_SELECT -> if (selecting) "End selection mode" else "Start selection mode"
        EditAction.COPY -> "Copy"
        EditAction.APPEND -> "Append, add to what was copied before"
        EditAction.CUT -> "Cut"
        EditAction.PASTE -> "Paste"
        EditAction.CLEAR -> "Clear edit box"
    }

    private fun stepAction(node: AccessibilityNodeInfo, dir: Int) {
        val list = actionsFor(node)
        if (actionIndex >= list.size) actionIndex = -1
        actionIndex = if (actionIndex < 0) {
            if (dir > 0) 0 else list.size - 1
        } else {
            wrap(actionIndex + dir, list.size)
        }
        playSound(soundFocus)
        speak(actionLabel(list[actionIndex]))
    }

    private fun runEditAction(node: AccessibilityNodeInfo, a: EditAction): OpResult = when (a) {
        EditAction.CURSOR_START -> moveCursorToEdge(node, false)
        EditAction.CURSOR_END -> moveCursorToEdge(node, true)
        EditAction.TOGGLE_SELECT -> toggleSelectionMode(node)
        EditAction.COPY -> opCopy(node, false)
        EditAction.APPEND -> opCopy(node, true)
        EditAction.CUT -> opCut(node)
        EditAction.PASTE -> opPaste(node)
        EditAction.CLEAR -> opClear(node)
    }

    private fun moveCursorToEdge(node: AccessibilityNodeInfo, toEnd: Boolean): OpResult {
        node.refresh()
        val len = editableText(node).length
        val target = if (toEnd) len else 0
        val ok: Boolean
        if (selecting) {
            val anchor = selAnchor.coerceIn(0, len)
            selCursor = target
            ok = setSelectionRange(node, minOf(anchor, target), maxOf(anchor, target))
        } else {
            ok = setSelectionRange(node, target, target)
        }
        if (!ok) return OpResult(false, "Could not move the cursor")
        val msg = if (selecting) {
            if (toEnd) "Selected to the end" else "Selected to the start"
        } else {
            if (toEnd) "Cursor at end" else "Cursor at start"
        }
        return OpResult(true, msg)
    }

    private fun toggleSelectionMode(node: AccessibilityNodeInfo): OpResult {
        if (selecting) {
            selecting = false
            return OpResult(true, "Selection mode ended")
        }
        node.refresh()
        val len = editableText(node).length
        val sel = selectionOf(node)
        if (sel != null) {
            selAnchor = sel.first
            selCursor = sel.second
        } else {
            val pos = cursorPos(node, len)
            selAnchor = pos
            selCursor = pos
        }
        selecting = true
        return OpResult(true, "Selection mode started. Now choose character, word, line or paragraph, then swipe down or up")
    }

    private fun speakPiece(piece: String, kind: SplitKind): String {
        return if (kind == SplitKind.CHAR) speakable(piece) else speakable(piece.trim().ifEmpty { piece })
    }

    // Moves the cursor of an edit box by one character/word/line/paragraph.
    // In selection mode the selection grows or shrinks between the anchor and the cursor.
    private fun cursorStep(node: AccessibilityNodeInfo, kind: SplitKind, dir: Int) {
        node.refresh()
        val text = editableText(node)
        if (text.isEmpty()) {
            fail("The edit box has no text")
            return
        }
        val from = if (selecting) selCursor.coerceIn(0, text.length) else cursorPos(node, text.length)
        val to = moveIndex(text, from, kind, dir)
        if (to == from) {
            fail(if (dir > 0) "End" else "Start")
            return
        }
        val piece = text.substring(minOf(from, to), maxOf(from, to))
        val ok: Boolean
        val message: String
        if (selecting) {
            val anchor = selAnchor.coerceIn(0, text.length)
            selCursor = to
            ok = setSelectionRange(node, minOf(anchor, to), maxOf(anchor, to))
            val extending = Math.abs(to - anchor) > Math.abs(from - anchor)
            message = speakPiece(piece, kind) + (if (extending) " selected" else " deselected")
        } else {
            ok = setSelectionRange(node, to, to)
            message = speakPiece(piece, kind)
        }
        if (!ok) {
            fail("Could not move the cursor")
            return
        }
        playSound(soundFocus)
        speak(message)
    }

    private fun moveIndex(text: String, pos: Int, kind: SplitKind, dir: Int): Int {
        val len = text.length
        return when (kind) {
            SplitKind.CHAR -> {
                val bi = BreakIterator.getCharacterInstance()
                bi.setText(text)
                if (dir > 0) {
                    if (pos >= len) {
                        len
                    } else {
                        val n = bi.following(pos)
                        if (n == BreakIterator.DONE) len else n
                    }
                } else {
                    if (pos <= 0) {
                        0
                    } else {
                        val n = bi.preceding(pos)
                        if (n == BreakIterator.DONE) 0 else n
                    }
                }
            }
            SplitKind.WORD -> {
                var p = pos
                if (dir > 0) {
                    while (p < len && text[p].isWhitespace()) p++
                    while (p < len && !text[p].isWhitespace()) p++
                } else {
                    while (p > 0 && text[p - 1].isWhitespace()) p--
                    while (p > 0 && !text[p - 1].isWhitespace()) p--
                }
                p
            }
            SplitKind.LINE -> {
                if (dir > 0) {
                    var p = pos
                    if (p < len && text[p] == '\n') p++
                    val idx = text.indexOf('\n', p)
                    if (idx == -1) len else idx
                } else {
                    var p = pos
                    if (p > 0 && text[p - 1] == '\n') p--
                    val idx = if (p <= 0) -1 else text.lastIndexOf('\n', p - 1)
                    idx + 1
                }
            }
            SplitKind.PARAGRAPH -> {
                val para = Regex("\\n\\s*\\n")
                if (dir > 0) {
                    val m = para.find(text, pos)
                    if (m == null) {
                        len
                    } else if (m.range.first > pos) {
                        m.range.first
                    } else {
                        val n = para.find(text, m.range.last + 1)
                        n?.range?.first ?: len
                    }
                } else {
                    var best = 0
                    for (m in para.findAll(text)) {
                        val st = m.range.last + 1
                        if (st < pos) best = st
                    }
                    best
                }
            }
        }
    }

    // ======================= Accessibility volume =======================

    private fun isInputMethodEvent(event: AccessibilityEvent): Boolean {
        return try {
            windows.any { it.id == event.windowId && it.type == AccessibilityWindowInfo.TYPE_INPUT_METHOD }
        } catch (e: Exception) {
            false
        }
    }

    // Volume keys pressed while a finger is held on the screen change the
    // accessibility (speech) volume; otherwise they change media volume as usual.
    override fun onKeyEvent(event: KeyEvent): Boolean {
        val code = event.keyCode
        if (gamingActive && (code == KeyEvent.KEYCODE_VOLUME_UP || code == KeyEvent.KEYCODE_VOLUME_DOWN)) {
            val down = event.action == KeyEvent.ACTION_DOWN
            if (code == KeyEvent.KEYCODE_VOLUME_UP) volUpHeld = down else volDownHeld = down
            if (volUpHeld && volDownHeld) {
                volUpHeld = false
                volDownHeld = false
                gamingSuppressedPackage = lastForegroundPackage
                exitGaming()
                return true
            }
            return false
        }
        if (touching && (code == KeyEvent.KEYCODE_VOLUME_UP || code == KeyEvent.KEYCODE_VOLUME_DOWN)) {
            if (event.action == KeyEvent.ACTION_DOWN) {
                adjustAccessibilityVolume(code == KeyEvent.KEYCODE_VOLUME_UP)
            }
            return true
        }
        return false
    }

    private fun adjustAccessibilityVolume(up: Boolean) {
        try {
            val am = getSystemService(Context.AUDIO_SERVICE) as AudioManager
            am.adjustStreamVolume(
                accessibilityStream,
                if (up) AudioManager.ADJUST_RAISE else AudioManager.ADJUST_LOWER,
                AudioManager.FLAG_SHOW_UI
            )
            val max = am.getStreamMaxVolume(accessibilityStream)
            val cur = am.getStreamVolume(accessibilityStream)
            val percent = if (max > 0) cur * 100 / max else 0
            speak("Accessibility volume " + percent + " percent")
        } catch (e: Exception) {
            fail("Could not change the accessibility volume")
        }
    }

    private fun startSelect() {
        if (targetText() == null) return
        mode = Mode.UNIT_CHOOSER
        unitIndex = 0
        playSound(soundFocus)
        speak("Which unit do you want to select by? " + unitLabels[0])
    }

    private fun splitText(text: String, kind: SplitKind, includeBlank: Boolean = false): List<String> {
        return when (kind) {
            SplitKind.CHAR -> {
                val out = mutableListOf<String>()
                val bi = BreakIterator.getCharacterInstance()
                bi.setText(text)
                var start = bi.first()
                var end = bi.next()
                while (end != BreakIterator.DONE) {
                    val piece = text.substring(start, end)
                    if (includeBlank || piece.isNotBlank()) out.add(piece)
                    start = end
                    end = bi.next()
                }
                out
            }
            SplitKind.WORD -> text.split(Regex("\\s+")).filter { it.isNotEmpty() }
            SplitKind.LINE -> text.split(Regex("\\r?\\n")).map { it.trim() }.filter { it.isNotEmpty() }
            SplitKind.PARAGRAPH -> text.split(Regex("\\r?\\n\\s*\\r?\\n")).map { it.trim() }.filter { it.isNotEmpty() }
        }
    }

    private fun beginItemSelect(kind: SplitKind) {
        val text = targetText() ?: return
        val units = splitText(text, kind)
        if (units.isEmpty()) {
            fail("Nothing found to select")
            return
        }
        splitKind = kind
        selectUnits = units
        selectedFlags = BooleanArray(units.size)
        selectIndex = 0
        mode = Mode.ITEM_SELECT
        playSound(soundActivate)
        speak(
            "" + units.size + " " + unitLabels[kind.ordinal] + "s found. Swipe left or right, " +
                "double tap to select or deselect, and at the end there are Copy and Append options. " +
                describeSelectEntry(0)
        )
    }

    private fun speakable(piece: String): String = when (piece) {
        " " -> "space"
        "\n", "\r\n" -> "new line"
        "\t" -> "tab"
        else -> piece
    }

    private fun describeSelectEntry(i: Int): String {
        val n = selectUnits.size
        return when {
            i < n -> speakable(selectUnits[i]) + ", " + (if (selectedFlags[i]) "selected" else "not selected")
            i == n -> "Copy"
            else -> "Append, add to what was copied before"
        }
    }

    private fun toggleSelected(i: Int) {
        selectedFlags[i] = !selectedFlags[i]
        playSound(soundActivate)
        speak(speakable(selectUnits[i]) + (if (selectedFlags[i]) " selected" else " deselected"))
    }

    private fun copySelection(append: Boolean) {
        val picked = selectUnits.filterIndexed { i, _ -> selectedFlags[i] }
        if (picked.isEmpty()) {
            fail("Nothing has been selected yet")
            return
        }
        val separator = when (splitKind) {
            SplitKind.CHAR -> ""
            SplitKind.WORD -> " "
            SplitKind.LINE -> "\n"
            SplitKind.PARAGRAPH -> "\n\n"
        }
        if (!append) clipList.clear()
        clipList.add(picked.joinToString(separator))
        if (writeClipboard()) {
            finish(
                if (append) "Added. The clipboard now has " + clipList.size + " items"
                else "" + picked.size + " selected parts copied"
            )
        } else {
            fail("Could not copy")
        }
    }

        // Required override - nothing to clean up mid-interrupt for v1.
    }

    override fun onDestroy() {
        prefsListener?.let {
            getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
                .unregisterOnSharedPreferenceChangeListener(it)
        }
        handler.removeCallbacksAndMessages(null)
        voiceStopRequested = true
        voiceExecutor.shutdown()
        aiExecutor.shutdown()
        tts?.shutdown()
        soundPool?.release()
        super.onDestroy()
    }

package com.mohit.screenreader

import android.content.Context
import android.media.AudioAttributes
import android.os.Bundle
import android.speech.tts.TextToSpeech
import android.widget.Button
import android.widget.TextView

/**
 * Text-to-speech settings: speed and volume of the speech engine.
 * Uses plain buttons (not sliders) so it works with double tap alone.
 * Values are saved in screen_reader_prefs; the running screen reader service
 * watches those and applies changes immediately.
 */
class TtsSettingsActivity : android.app.Activity() {

    private var tts: TextToSpeech? = null
    private var ttsReady = false
    private lateinit var txtRate: TextView
    private lateinit var txtVolume: TextView
    private var rate = 1.0f
    private var volume = 1.0f

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_tts_settings)
        txtRate = findViewById(R.id.txtRate)
        txtVolume = findViewById(R.id.txtVolume)

        val prefs = getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
        rate = prefs.getFloat("tts_rate", 1.0f)
        volume = prefs.getFloat("tts_volume", 1.0f)
        val enginePackage = prefs.getString("tts_engine_package", null)
        showValues()

        val initListener = TextToSpeech.OnInitListener { status ->
            if (status == TextToSpeech.SUCCESS) {
                tts?.setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_ASSISTANCE_ACCESSIBILITY)
                        .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                        .build()
                )
                ttsReady = true
            }
        }
        tts = if (enginePackage != null) {
            TextToSpeech(this, initListener, enginePackage)
        } else {
            TextToSpeech(this, initListener)
        }

        findViewById<Button>(R.id.btnRateUp).setOnClickListener { changeRate(0.25f) }
        findViewById<Button>(R.id.btnRateDown).setOnClickListener { changeRate(-0.25f) }
        findViewById<Button>(R.id.btnVolUp).setOnClickListener { changeVolume(0.1f) }
        findViewById<Button>(R.id.btnVolDown).setOnClickListener { changeVolume(-0.1f) }
        findViewById<Button>(R.id.btnTest).setOnClickListener {
            speakTest("This is your text-to-speech voice")
        }
    }

    private fun rateText(): String = "Speed: " + rate + " times"

    private fun volumeText(): String = "Text to speech volume: " + Math.round(volume * 100) + " percent"

    private fun showValues() {
        txtRate.text = rateText()
        txtVolume.text = volumeText()
    }

    private fun save() {
        getSharedPreferences("screen_reader_prefs", Context.MODE_PRIVATE)
            .edit()
            .putFloat("tts_rate", rate)
            .putFloat("tts_volume", volume)
            .apply()
    }

    private fun changeRate(delta: Float) {
        rate = Math.round((rate + delta) * 100) / 100f
        if (rate < 0.5f) rate = 0.5f
        if (rate > 3.0f) rate = 3.0f
        save()
        showValues()
        speakTest(rateText())
    }

    private fun changeVolume(delta: Float) {
        volume = Math.round((volume + delta) * 10) / 10f
        if (volume < 0.1f) volume = 0.1f
        if (volume > 1.0f) volume = 1.0f
        save()
        showValues()
        speakTest(volumeText())
    }

    // Speaks with the current speed and volume so the change can be heard right away.
    private fun speakTest(text: String) {
        val engine = tts
        if (engine == null || !ttsReady) return
        engine.setSpeechRate(rate)
        val params = Bundle()
        params.putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, volume)
        engine.speak(text, TextToSpeech.QUEUE_FLUSH, params, "tts_settings_test")
    }

    override fun onDestroy() {
        tts?.shutdown()
        super.onDestroy()
    }
}
